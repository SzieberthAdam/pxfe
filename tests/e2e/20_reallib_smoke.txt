# 20. Smoke test on the real library (read-only use): the browser opens on pixelglyphlib (the folder or its pxfdb.pxfdb), 5 different fonts open locked.
# Not part of run_all.bat (slow): run tests\e2e\run_reallib.bat.  The database that was loaded is never saved; Save As writes a copy under tests/tmp.
# The fonts are named by their hashes, so they are fixed points of the library (see README.md).
waitfor loading = 0 900000
wait 10
assert inbrowser 1
assert hasstore 1
waitfor btotal >= 1000 60000
waitfor bpending = 0 60000
assert btab 0
shot tests/tmp/shots/20_real_browser.png

# ---- 1. a FON typeface font with RATIO (ProgTinyCo)
click @Typefaces
wait 4
click @Filter
key a ctrl
type ProgTinyCo
wait 6
assert bvisible >= 1
click @card:0
wait 3
dblclick @card:0
waitfor inbrowser = 0 120000
assert readonly 1
assert modified 0
assert hashname 1
assert hasratio 1
assert hascont 1
click @Unlock
wait 10
assert readonly 0
assert modified 0
assert unsaved 1
assert dbmodified 0
assert hashname 1
shot tests/tmp/shots/20_real_1.png
click @nav.0
wait 8
click @Typefaces
wait 6
assert inbrowser 1
