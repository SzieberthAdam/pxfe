# 42. Adding layers in the glyph view of a duplicate of 3b52125b (glyphs 20 2C 41 48 4D 78): from this font (same code, a typed
# character, a U+ code, a code the font lacks), from another font of the library (Test Sans Italic 4fba4190), the layers following
# PgDn / PgUp, and staying while another font (Uncoded Mono 54caf68f) is opened.  Layers never touch the font.
wait 5
click @Unlock
wait 4
assert hasstore 1
assert readonly 0
click @cell:U+0041
key enter
wait 6
assert editor 1
click @gtab_Layers
wait 3
assert l.count 1
assert l.cur 0
assert l.sel 0
assert ehidden 0

# Add Layer: the popover opens; this font, the same code as the edited glyph
click @+_Add_Layer
wait 3
assert l.pop 1
assert w.lay_pop 1
shot tests/tmp/shots/42_popover_this.png
click @Add
wait 3
assert l.pop 0
assert l.count 2
assert l.cur 1
assert l.sel 0
assert l.this.0 1
assert l.same.0 1
assert l.code.0 U+0041
assert l.ink.0 > 0

# a typed character (the "same code" choice is remembered: switch it off)
click @+_Add_Layer
wait 3
click @lay_same
click @lay_glyph
type x
wait 2
click @Add
wait 3
assert l.count 3
assert l.cur 2
assert l.same.0 0
assert l.code.0 'x
assert l.ink.0 > 0

# a U+ code

wait 3

assert l.ink.0 > 0
