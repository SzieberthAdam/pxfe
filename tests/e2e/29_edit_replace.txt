# 29. An unlocked font is committed into the store, in place, when its font view is left; a later commit replaces the earlier one (no
# in-between fonts), and editing back to the original content gives the original hash again.
wait 5
click @Unlock
wait 4
assert readonly 0
assert sql "select count(*) from pxfm" 4
assert sql "select count(*) from pxtf" 2
click @ascent+
wait 6
click @nav.0
wait 8
click @Typefaces
wait 8
assert dbmodified 1
assert sql "select count(*) from pxfm" 4
assert sql "select count(*) from pxtf" 2
assert sql "select count(*) from pxfm where ascent = 7" 1
# back to the font, another change, leave again: the first commit is replaced
key right alt
wait 6
assert infont 1
assert unsaved 1
click @ascent+
wait 6
click @nav.0
wait 8
click @Typefaces
wait 8
assert sql "select count(*) from pxfm" 4
assert sql "select count(*) from pxtf" 2
assert sql "select count(*) from pxfm where ascent = 7" 0
assert sql "select count(*) from pxfm where ascent = 8" 1
# leaving without a change commits nothing more
key right alt
wait 6
click @nav.0
wait 8
click @Typefaces
wait 8
assert sql "select count(*) from pxfm" 4
# three edits in a row, leaving after each: still one font
key right alt
wait 6
click @ascent+
wait 6
click @nav.0
wait 8
click @Typefaces
wait 8
key right alt
wait 6
click @ascent+
wait 6
click @nav.0
wait 8
click @Typefaces
wait 8
assert sql "select count(*) from pxfm" 4
assert sql "select count(*) from pxfm where ascent = 10" 1
assert sql "select count(*) from pxtf" 2
# leaving through the Glyphs tab commits too
key right alt
wait 6
click @ascent+
wait 6
click @Glyphs
wait 8
assert sql "select count(*) from pxfm" 4
assert sql "select count(*) from pxfm where ascent = 11" 1
# edited back to the original: the commit goes away, the database is as loaded
key right alt
wait 6
click @ascent-
wait 6
click @ascent-
wait 6
click @ascent-
wait 6
click @ascent-
wait 6
click @ascent-
wait 6
assert modified 1
click @nav.0
wait 8
click @Typefaces
wait 8
assert sql "select count(*) from pxfm" 4
assert sql "select count(*) from pxtf" 2
assert dbmodified 1
assert sql "select count(*) from pxfm where lower(hex(hash)) = '3b52125b2e5f6466823baa4fa3186fa0'" 1
assert sql "select count(*) from pxtf_variant where lower(hex(pxfm)) = '3b52125b2e5f6466823baa4fa3186fa0'" 1
