# 139. Version 22: per-glyph palettes (version 23: the palette bar, canonical slots).  Database testdata/lib_many.pxfdb (made by tests\make_testdata.exe), font b6a511be of COLORS 40, unlocked.
# The font has 40 colours, more than one glyph may have (36): A..T have two colours each (A: 20F040 25EB70, B: 2AE6A0 2FE1D0, ...), U and V share
# A's palette, W (6x6) has 36 colours, one per pixel in reading order (c0 = 20F040 .. c35).
# Pixel rows are rows of the font box: A, B and U have their top row at row 1, W at row 0.
# Drawing a colour adds it to that glyph only; the eyedropper picks a colour; a 37th colour in W is refused with a toast, and taken once a
# colour of W is erased; a paste into a glyph of another palette brings its colours along; saved, the font keeps COLORS 40.
wait 5
click @Unlock
wait 4
assert glyphs 24
assert colors 40
assert palette 40
assert palettes 22
sleep 600

# A: two colours; the ink becomes the first colour of the font
dblclick @cell:U+0041
wait 5
assert editor_glyph U+0041
assert epal 2
assert ecolors 2
assert einkrgb 0x20F040
assert glyph U+0041 color 0 2 20F040
assert glyph U+0041 color 0 4 25EB70
assert glyph U+0041 color 1 2 none

# drawing B's first colour into A: A gets a palette of three (copy on write); U, which shares A's palette, and B keep theirs
click @pal:#2AE6A0
wait 2
assert einkrgb 0x2AE6A0
click @ecell:1,2
wait 3
assert glyph U+0041 color 1 2 2AE6A0
assert glyph U+0041 color 0 2 20F040
assert epal 3
assert ecolors 3
assert glyph U+0055 color 0 1 20F040
assert glyph U+0055 color 1 5 25EB70
assert glyph U+0042 color 0 2 2AE6A0
assert glyph U+0042 color 0 4 2FE1D0
assert colors 40
assert palettes 23

# the eyedropper picks the colour under the cursor; Alt with the Pencil does the same
key i shift
wait 2
click @ecell:0,4
wait 3
assert einkrgb 0x25EB70
# version 23: A is canonical, so the colour drawn at 1,2 (before 25EB70 in reading order) is slot 1, 25EB70 slot 2
assert eink '2
click @ecell:1,2
wait 3
assert einkrgb 0x2AE6A0
key b shift
wait 2
click @ecell:0,2 alt
wait 3
assert einkrgb 0x20F040
assert undo 1

# copy A's top row, paste it into B: B's palette gains A's colour
key m shift
wait 2
drag @ecell:0,1 @ecell:3,1
wait 3
key c ctrl
wait 2
click @Back
wait 4
sleep 600
dblclick @cell:U+0042
wait 5
assert editor_glyph U+0042
assert epal 2
key v ctrl
wait 3
assert efloating 1
drag @ecell:0,1 @ecell:0,2
wait 3
key enter
wait 3
assert efloating 0
assert glyph U+0042 color 0 2 20F040
assert glyph U+0042 color 3 2 20F040
assert glyph U+0042 color 0 1 2AE6A0
assert glyph U+0042 color 0 4 2FE1D0
assert epal 3
assert ecolors 3
assert glyph U+0041 color 0 2 20F040
key d ctrl
key b shift
wait 2

# W has 36 colours: a 37th (S's first, D43C40, in the bar's row of the font's other colours) is refused
click @Back
wait 4
sleep 600
dblclick @cell:U+0057
wait 5
assert editor_glyph U+0057
assert epal 36
assert ecolors 36
assert glyph U+0057 color 0 0 20F040
assert glyph U+0057 color 1 0 25EB70
assert glyph U+0057 color 5 5 CF41D0
assert ebarrows 2
assert ebarextras 4
click @pal:#D43C40
wait 3
assert einkrgb 0xD43C40
assert editor 1
click @ecell:2,2
wait 3
assert toast has 36 colours
assert glyph U+0057 color 2 2 66AAA0
assert ecolors 36
# the right button erases ('.'): the erased colour's slot is free, so the new colour takes it
click @ecell:1,0 right
wait 3
assert glyph U+0057 color 1 0 none
assert ecolors 35
click @ecell:1,0
wait 3
assert glyph U+0057 color 1 0 D43C40
assert epal 36
assert ecolors 36
assert colors 40

# saved: a font of COLORS 40 that is not the one opened
key esc
wait 3
assert editor 0
key s ctrl
waitfor unsaved == 0 20000
assert sql "SELECT COUNT(*) FROM pxfm WHERE colors = 40 AND hash <> 'b6a511befd442427c26d8d6dc6033a90'" 1
assert sql "SELECT COUNT(*) FROM pxfm_glyph WHERE glyph NOTNULL AND palette ISNULL" 0
assert sql "SELECT COUNT(*) FROM pxfm_glyph WHERE palette NOT IN(SELECT hash FROM palette)" 0
