# 26. Export, Append: a folder that has some of the files and other files.  The dialog shows what is found; Append writes only what is
# missing, never changes an existing file, never deletes.  The duplicate being edited is committed first and exported too.
wait 5
click @Unlock
wait 4
assert readonly 0
writefile tests/tmp/xout/pxfm/3b/3b52125b2e5f6466823baa4fa3186fa0.pxfm SENTINEL\n
writefile tests/tmp/xout/notes.txt keep me\n
writefile tests/tmp/xout/other/deep.txt keep me too\n
click @ascent+
wait 6
assert unsaved 1
setdialog folder tests/tmp/xout
click @Menu
wait 3
click @file.export
wait 8
assert xdopen 1
assert xfound_fonts 1
assert xfound_typefaces 0
assert xbackup 1
shot tests/tmp/shots/v5_export_dialog.png
click @Append
wait 12
assert xdopen 0
assert modal Export_finished
assert xw_fonts 4
assert xw_typefaces 2
assert xs_fonts 0
assert xd_fonts 0
key enter
wait 4
assert modal none
# the existing file is as it was, the others were added, nothing else was touched
assert file tests/tmp/xout/pxfm/3b/3b52125b2e5f6466823baa4fa3186fa0.pxfm contains SENTINEL
assert file tests/tmp/xout/notes.txt contains keep me
assert file tests/tmp/xout/other/deep.txt contains keep me too
assert samefile tests/tmp/xout/pxfm/4f/4fba4190f6ba7e10b296c7070234e1ce.pxfm tests/tmp/e2elib/pxfm/4f/4fba4190f6ba7e10b296c7070234e1ce.pxfm
assert file $SAVEDPXFM exists
assert file $SAVEDPXTF exists
# Append does not make a backup and does not change the database file
assert samefile tests/tmp/e2e.pxfdb testdata/lib.pxfdb
assert dbmodified 1
