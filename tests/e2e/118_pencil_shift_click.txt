# 118. Shift+click with the Pencil (version 17), font 0820dc69 unlocked, glyph U+0020 (empty, advance 4; font box rows 0..7).
# Shift+click draws a line from the last cell the pencil (or the line tool) ended on; the right button does it with '.'; one undo step each.
wait 5
click @Unlock
wait 4
sleep 600
dblclick @cell:U+0020
wait 5
assert editor_glyph U+0020
assert glyph U+0020 empty
assert etool 0

# nothing to start from yet: Shift+click is an ordinary click
click @ecell:1,1 shift
wait 3
assert glyph U+0020 ink 1
assert undo 1

# now it is a line from (1,1) to (6,1)
click @ecell:6,1 shift
wait 3
assert glyph U+0020 ink 6
assert glyph U+0020 pixel 3 1 0
assert glyph U+0020 pixel 6 1 0
assert undo 2

# and on from there, to (6,6): a vertical line
click @ecell:6,6 shift
wait 3
assert glyph U+0020 ink 11
assert glyph U+0020 pixel 6 4 0
assert undo 3
shot tests/tmp/shots/118_shift_click.png

# the stroke is one step: a drag after the Shift+click continues as freehand
click @ecell:0,7
wait 3
assert glyph U+0020 ink 12
assert undo 4

# the last cell is now (0,7): Shift+click draws to (3,4), a diagonal
click @ecell:3,4 shift
wait 3
assert glyph U+0020 pixel 1 6 0
assert glyph U+0020 pixel 2 5 0
assert glyph U+0020 pixel 3 4 0
assert glyph U+0020 ink 15
assert undo 5

# undo takes the line away in one step
key z ctrl
wait 3
assert undo 4
assert glyph U+0020 ink 12
key z ctrl
key z ctrl
key z ctrl
key z ctrl
wait 3
assert undo 0
assert glyph U+0020 empty

# the right button with Shift draws '.': first a row, then a gap cut through it
drag @ecell:0,3 @ecell:6,3
wait 3
assert glyph U+0020 ink 7
click @ecell:1,3 right
wait 3
assert glyph U+0020 ink 6
click @ecell:4,3 right shift
wait 3
assert glyph U+0020 ink 3
assert glyph U+0020 pixel 1 3 .
assert glyph U+0020 pixel 3 3 .
assert glyph U+0020 pixel 4 3 .
assert glyph U+0020 pixel 0 3 0
assert glyph U+0020 pixel 6 3 0
assert undo 3
key z ctrl
key z ctrl
key z ctrl
wait 3
assert glyph U+0020 empty

# the Line tool's end is the start of the next Shift+click with the pencil
key l
drag @ecell:1,1 @ecell:1,5
wait 3
assert glyph U+0020 ink 5
key b
wait 2
click @ecell:5,5 shift
wait 3
assert glyph U+0020 ink 9
assert glyph U+0020 pixel 3 5 0
assert glyph U+0020 pixel 5 5 0
key z ctrl
key z ctrl
wait 3
assert undo 0
assert glyph U+0020 empty

# the glyph view opened again has no last cell: Shift+click is a click
click @ecell:2,2
wait 2
key esc
wait 4
assert editor 0
dblclick @cell:U+0020
wait 5
assert editor 1
click @ecell:5,5 shift
wait 3
assert glyph U+0020 ink 2
