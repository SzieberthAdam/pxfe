# 98. A locked font may get font layers: they are program settings, the .pxfdb stays byte for byte as it was (assert
# samefile against a copy), the sidecar is written, and the layers survive Unlock.  Font 3b52125b.
wait 5
assert readonly 1
assert flayers 0
copyfile tests/tmp/e2e.pxfdb tests/tmp/e2e_before.pxfdb
click @cell:U+0041
key enter
wait 6
assert editor 1
assert readonly 1
click @gtab_Layers
wait 3
click @+_Add_Layer
wait 3
click @lay_src_File
wait 3
setdialog layerfile testdata/outline/NotoSub.ttf
click @lay_choose
wait 6
click @Add
wait 6
assert flayers 1
assert readonly 1
# edit it in the original: colour, offset, a correction for A
click @lay_hex
type #cc2288
key enter
wait 3
click @lay_x+
click @lay_x+
click @lay_gx+
click @lay_gy+
wait 3
assert flayer 0 color 0xCC2288
assert flayer 0 dx 0.5
assert fcorr 0 U+0041 exists 1
assert modified 0
assert unsaved 1
shot tests/tmp/fl/orig_layer.png
# Save: only the sidecar is written
key s ctrl
wait 12
assert unsaved 0
assert sidecar exists
assert samefile tests/tmp/e2e.pxfdb tests/tmp/e2e_before.pxfdb
assert sqlfile tests/tmp/e2e.pxfdb-settings "select count(*) from layer where font = '3b52125b2e5f6466823baa4fa3186fa0'" 1
assert sqlfile tests/tmp/e2e.pxfdb-settings "select count(*) from correction" 1
# Unlock keeps the layers and the correction
click @Unlock
wait 8
assert readonly 0
assert flayers 1
assert flayer 0 color 0xCC2288
assert fcorr 0 U+0041 dx 0.25
assert l.count 2
# editing a pixel moves them to the new hash; nothing stays under the old one
click @ecell:2,3
wait 4
assert modified 1
wait 6
assert flayers 1
key escape
wait 6
key s ctrl
wait 12
assert sqlfile tests/tmp/e2e.pxfdb-settings "select count(*) from layer" 1
assert sqlfile tests/tmp/e2e.pxfdb-settings "select count(*) from layer where font = '3b52125b2e5f6466823baa4fa3186fa0'" 0
assert sqlfile tests/tmp/e2e.pxfdb-settings "select count(*) from correction" 1
assert sqlfile tests/tmp/e2e.pxfdb "select count(*) from pxfm" 4
