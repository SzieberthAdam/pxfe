# 115. Delete, cut, copy and paste of a selection (version 16), font 0820dc69 unlocked, glyphs U+0020 (empty, advance 4) and U+0078.
# Delete and cut clear the selected cells (one undo step each); the clipboard keeps the pixels and the cells; a paste floats where the pixels were
# copied from (or at the top left of the view when they would be out of view) and is one undo step; '.' cells of the pixels are transparent.
wait 5
click @Unlock
wait 4
sleep 600
dblclick @cell:U+0020
wait 5
assert editor_glyph U+0020
assert eclip 0

# two rows of ink: columns 1..3 of rows 1 and 2
drag @ecell:1,1 @ecell:3,1
drag @ecell:1,2 @ecell:3,2
wait 3
assert glyph U+0020 ink 6
assert undo 2
key m shift
wait 2

# Delete clears the selected cells; one undo step; the bitmap is cropped to what is left
drag @ecell:0,0 @ecell:4,1
wait 3
assert eselcount 10
key delete
wait 3
assert glyph U+0020 ink 3
assert glyph U+0020 pixel 1 1 .
assert glyph U+0020 pixel 1 2 0
assert glyph U+0020 top 2
assert glyph U+0020 size 3x1
assert undo 3
assert esel bounds 0 0 5 2
# Backspace is the same; nothing selected, nothing changes; selected empty cells change nothing
key backspace
wait 2
assert undo 3
key d ctrl
key backspace
wait 2
assert undo 3
key z ctrl
wait 3
assert glyph U+0020 ink 6
assert undo 2

# Cut: the pixels go to the clipboard, the cells are cleared
drag @ecell:0,0 @ecell:4,1
wait 3
key x ctrl
wait 3
assert eclip 10
assert glyph U+0020 ink 3
assert undo 3
# Paste: the pixels float where they were; the marquee is the tool; nothing is written yet
key v ctrl
wait 3
assert efloating 1
assert esel bounds 0 0 5 2
assert etool 5
assert glyph U+0020 ink 3
assert undo 3
key enter
wait 3
assert efloating 0
assert glyph U+0020 ink 6
assert glyph U+0020 pixel 1 1 0
assert undo 4
shot tests/tmp/shots/115_pasted.png

# Copy leaves the glyph alone; a paste moved by a drag lands there; the '.' cells of the pixels do not erase what they land on
key d ctrl
drag @ecell:0,0 @ecell:4,1
wait 3
key c ctrl
wait 2
assert eclip 10
assert undo 4
key v ctrl
wait 3
assert efloating 1
drag @ecell:2,1 @ecell:2,2
wait 3
assert esel bounds 0 1 5 2
key enter
wait 3
assert undo 5
assert glyph U+0020 pixel 1 1 0
assert glyph U+0020 pixel 1 2 0
assert glyph U+0020 pixel 3 2 0
assert glyph U+0020 ink 6
# a paste is thrown away by Esc and by Ctrl+Z, and writes nothing
key d ctrl
key v ctrl
wait 2
assert efloating 1
key esc
wait 2
assert efloating 0
assert esel none
assert undo 5
key v ctrl
wait 2
assert efloating 1
key z ctrl
wait 3
assert efloating 0
assert undo 5
assert editor 1

# a paste does not fit in the view when the view has been moved away: it goes to the top left of the view
drag 300 400 1000 400 middle
wait 3
key v ctrl
wait 3
assert efloating 1
assert esel bounds * * 5 2
assert esel cell 0 0 0
key esc
wait 2
assert undo 5

# the clipboard survives the glyph: paste into U+0078 (a new glyph view, no selection)
click @Back
wait 4
sleep 600
dblclick @cell:U+0078
wait 5
assert editor_glyph U+0078
assert esel none
assert eclip 10
key v ctrl
wait 3
assert efloating 1
assert esel bounds 0 0 5 2
key enter
wait 3
assert undo 6
assert glyph U+0078 pixel 1 1 0
assert glyph U+0078 pixel 3 1 0
key z ctrl
wait 3
assert undo 5
assert glyph U+0078 pixel 1 1 .
# leaving the glyph drops a floating paste: Back
key v ctrl
wait 3
assert efloating 1
click @Back
wait 4
assert editor 0
assert undo 6
assert glyph U+0078 pixel 1 1 0
