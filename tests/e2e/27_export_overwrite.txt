# 27. Export, Overwrite with a backup: the folder's current data is imported into a backup .pxfdb first, then the contents of
# pxtf/ pxfm/ pxgl/ are deleted and memory is written there.  Files outside those folders survive.
waitfor loading = 0 60000
wait 5
setdialog folder tests/tmp/xout
click @Menu
wait 3
click @file.export
wait 10
assert modal Export_finished
key enter
wait 4
writefile tests/tmp/xout/keep.txt keep me\n
writefile tests/tmp/xout/other/keep2.txt keep me too\n
delete tests/tmp/xout_backup.pxfdb
# delete a font in memory (the one without a typeface)
click @Fonts
wait 4
click @Filter
type 894ef8
wait 4
assert bvisible 1
sleep 600
click @card:0
wait 3
key delete
wait 5
assert modal Delete_font
click @Delete
wait 6
assert modal none
assert storefonts 3
assert dbmodified 1
# the dialog: what is found, the backup option (default on), the backup path through a Save dialog
click @Menu
wait 3
click @file.export
wait 8
assert xdopen 1
assert xfound_fonts 4
assert xfound_typefaces 2
assert xfound_glyphs 6
assert xbackup 1
setdialog save tests/tmp/xout_backup.pxfdb
click @xd.change
wait 4
shot tests/tmp/shots/v5_export_overwrite.png
click @Overwrite
wait 20
assert xdopen 0
assert modal Export_finished
assert xd_fonts 4
assert xd_typefaces 2
assert xd_glyphs 6
assert xw_fonts 3
assert xw_typefaces 2
key enter
wait 4
assert modal none
# the backup holds the old data
assert file tests/tmp/xout_backup.pxfdb exists
assert sqlfile tests/tmp/xout_backup.pxfdb "select count(*) from pxfm" 4
assert sqlfile tests/tmp/xout_backup.pxfdb "select count(*) from pxtf" 2
assert sqlfile tests/tmp/xout_backup.pxfdb "select count(*) from pxfm where lower(hex(hash)) = '894ef821e147533c34b39e2131016133'" 1
# the folder equals memory; other files survive
assert file tests/tmp/xout/pxfm/89/894ef821e147533c34b39e2131016133.pxfm missing
assert file tests/tmp/xout/palette/eb/ebbba94934da6897d86c5e8ea2ac5409.hex missing
assert file tests/tmp/xout/pxfm/3b/3b52125b2e5f6466823baa4fa3186fa0.pxfm exists
assert file tests/tmp/xout/pxfm/4f/4fba4190f6ba7e10b296c7070234e1ce.pxfm exists
assert file tests/tmp/xout/pxtf/ff/ff6354ce745de4d0170855e161a491be.pxtf exists
assert file tests/tmp/xout/keep.txt contains keep me
assert file tests/tmp/xout/other/keep2.txt contains keep me too
# the database file is still as it was loaded; memory still differs from it
assert samefile tests/tmp/e2e.pxfdb testdata/lib.pxfdb
assert dbmodified 1
