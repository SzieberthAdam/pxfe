# 125. The Rectangle tool (version 19), font 3b52125b unlocked, glyph U+0020 (empty, advance 4; font box rows 0..7, @ecell:x,y).
# A drag draws the outline from corner to corner in one undo step; Shift makes a square, Ctrl draws from the centre, both together a square from the centre;
# the modifiers count at the release; the bar's Filled toggle draws it solid; Esc cancels; the right button draws '.'.
wait 5
click @Unlock
wait 4
sleep 600
dblclick @cell:U+0020
wait 5
assert editor_glyph U+0020
assert glyph U+0020 empty
assert undo 0
assert efilled 0

key r shift
wait 3
assert etool 3
assert w.Filled 1
assert w.Pixel-perfect 0

# the outline: 4 x 3 cells, 10 of them, the inside stays empty; one undo step
drag @ecell:1,1 @ecell:4,3
wait 3
assert glyph U+0020 ink 10
assert glyph U+0020 pixel 1 1 0
assert glyph U+0020 pixel 4 3 0
assert glyph U+0020 pixel 4 1 0
assert glyph U+0020 pixel 1 3 0
assert glyph U+0020 pixel 2 2 .
assert glyph U+0020 pixel 3 2 .
assert undo 1
key z ctrl
wait 3
assert glyph U+0020 empty

# dragged up and to the left: the same outline
drag @ecell:4,3 @ecell:1,1
wait 3
assert glyph U+0020 ink 10
assert glyph U+0020 pixel 1 1 0
key z ctrl
wait 3

# Shift: a square of the longer side (4 across, 2 down: 5 x 5 cells, 16 of them)

wait 3

assert glyph U+0020 pixel 5 5 0
assert glyph U+0020 pixel 1 5 0
assert glyph U+0020 pixel 3 3 .
assert glyph U+0020 pixel 6 5 .

wait 3

key z ctrl
