# 31. Two duplicates of the same font edited to the same result (the same hash): leaving one and editing it further must not remove
# the font the other one committed, and a duplicate edited back to exactly its original leaves the original alone.
wait 5
assert readonly 0
assert sql "select count(*) from pxfm" 4
click @ascent.plus
wait 6
click @Library
wait 8
assert sql "select count(*) from pxfm" 5
assert sql "select count(*) from pxfm where ascent = 7" 1
# a second duplicate of the same original, edited to the same result
click @card:0 right
wait 3
click @Ctx_Duplicate_to_Edit
wait 8
assert blastpxfm 0820dc69a21b1cdb3a44d2cff7d884d8
assert readonly 0
click @ascent.plus
wait 6
click @Library
wait 8
assert sql "select count(*) from pxfm" 5
assert sql "select count(*) from pxfm where ascent = 7" 1
# the second one edited further: the shared font stays, the second one's own result is added
click @tab.font
wait 6
click @ascent.plus
wait 6
click @Library
wait 8
assert sql "select count(*) from pxfm" 6
assert sql "select count(*) from pxfm where ascent = 7" 1
assert sql "select count(*) from pxfm where ascent = 8" 1
