# 17. Save: duplicate a font of a typeface, edit it, Save.  The file's bytes change only at Save; the new rows are in the file.
waitfor loading = 0 60000
wait 5
assert storefonts 4
assert storetypefaces 2
copyfile tests/tmp/e2e.pxfdb tests/tmp/e2e_before.pxfdb
click @Fonts
wait 4
click @Filter
type 0820dc
wait 4
assert bvisible 1
click @card:0
wait 3
dblclick @card:0
wait 6
assert readonly 1
assert hashname 1
click @Duplicate_to_Edit
wait 8
assert readonly 0
assert modified 0
assert hascont 1
assert variant 1
assert unsaved 0
# an edit: ascent, and a stroke in the space glyph
click @ascent+
wait 6
click @cell:U+0020
key enter
wait 5
click @ecell:2,3
wait 3
click @Back
wait 4
assert modified 1
assert unsaved 1
assert wtitle modified
# leaving the font view commits the duplicate into memory; the file is still as it was
click @nav.0
wait 8
click @Typefaces
wait 8
assert dbmodified 1
assert wtitle modified
assert sql "SELECT COUNT(*) FROM pxfm" 5
assert sql "SELECT COUNT(*) FROM pxtf" 3
assert samefile tests/tmp/e2e.pxfdb tests/tmp/e2e_before.pxfdb
# Save button
click @Save
wait 12
assert saved pxfm
assert saved pxtf
assert toast Saved e2e.pxfdb
assert dbmodified 0
assert unsaved 0
assert statusdb saved
assert snapdir unchanged
assert sql "SELECT ascent FROM pxfm ORDER BY ascent DESC LIMIT 1" 7
assert sqlfile tests/tmp/e2e.pxfdb "SELECT COUNT(*) FROM pxfm" 5
assert sqlfile tests/tmp/e2e.pxfdb "SELECT COUNT(*) FROM pxtf" 3
assert sqlfile tests/tmp/e2e.pxfdb "SELECT COUNT(*) FROM pxfm WHERE ascent = 7" 1
# no temp file is left behind
assert file tests/tmp/e2e.pxfdb.tmp missing
