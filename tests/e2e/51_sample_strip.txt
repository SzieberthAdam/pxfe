# 51. The font view's sample strip (the read-only original F1; pxfe.ini is kept for 53): the sample, the zoom,
# the multi-line editor, pixels of both lines, resizing and collapsing.
wait 8
assert inbrowser 0
assert stripzoom 2
assert striph 140
assert stripcollapsed 0
assert samplelines 1

# ---- two lines of H in the strip; pixels of both lines at 3x (the H of this font: 5x5 px, stem on the left, box 6 + 2 rows, line gap 1)
click @sz+
wait 2
assert stripzoom 3
click @Sample_Edit
key a ctrl
type H
key enter
type H
key enter ctrl
wait 3
assert sample H\nH
assert samplelines 2
# first line: the stem of the H (the glyph starts one row below the top of the box)
assert pixel @Sample_Origin 1 4 141418
# right of the stem: the background of the preview
assert pixel @Sample_Origin 7 4 F8F8FA
# second line: its box starts 9 rows lower
assert pixel @Sample_Origin 1 31 141418
# the gap rows between two lines are empty
assert pixel @Sample_Origin 1 23 F8F8FA
shot tests/tmp/shots/51_strip.png

# ---- zoom 1x to 8x
repeat 2 click @sz-
wait 2
assert stripzoom 1
assert pixel @Sample_Origin 0 1 141418
repeat 9 click @sz+
wait 2
assert stripzoom 8
repeat 3 click @sz-
wait 2
assert stripzoom 5

# ---- resize by dragging the top edge: up to the top of the grid, then back down to the middle of the strip
drag @Sample_Resize @grid.scrollbar
wait 3
assert striph > 140
assert striph < 900
drag @Sample_Resize @Sample_Preview
wait 3
assert striph >= 96

# ---- collapse and expand with the chevron
click @Sample_Fold
wait 3
assert stripcollapsed 1
click @Sample_Fold
wait 3
assert stripcollapsed 0
click @Sample_Fold
wait 3
assert stripcollapsed 1
sleep 300
assert file tests/tmp/e2e.ini contains stripCollapsed=1
assert file tests/tmp/e2e.ini contains stripZoom=5
