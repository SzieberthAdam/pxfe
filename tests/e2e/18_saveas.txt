# 18. Save As: the edited database goes to a new file; the old file is not changed; later Saves go to the new file.
waitfor loading = 0 60000
wait 5
copyfile tests/tmp/e2e.pxfdb tests/tmp/e2e_before.pxfdb
click @Fonts
wait 4
click @Filter
type 0820dc
wait 4
click @card:0
wait 3
dblclick @card:0
wait 6
click @Unlock
wait 8
click @ascent+
wait 6
assert unsaved 1
# Save As to a new name (Ctrl+Shift+S, the dialog is overridden)
setdialog save tests/tmp/e2e_copy.pxfdb
key s ctrl shift
wait 12
assert modal none
assert toast Saved as e2e_copy.pxfdb
assert wtitle e2e_copy.pxfdb
assert statusdb e2e_copy.pxfdb
assert statusdb saved
assert unsaved 0
assert samefile tests/tmp/e2e.pxfdb tests/tmp/e2e_before.pxfdb
assert sqlfile tests/tmp/e2e_copy.pxfdb "SELECT COUNT(*) FROM pxfm" 4
assert sqlfile tests/tmp/e2e_copy.pxfdb "SELECT COUNT(*) FROM pxfm WHERE ascent = 7" 1
assert sqlfile tests/tmp/e2e.pxfdb "SELECT COUNT(*) FROM pxfm" 4
# another change, plain Save: goes to the copy
click @ptab_Font
click @ascent+
wait 6
assert unsaved 1
key s ctrl
wait 12
assert unsaved 0
assert samefile tests/tmp/e2e.pxfdb tests/tmp/e2e_before.pxfdb
assert sqlfile tests/tmp/e2e_copy.pxfdb "SELECT COUNT(*) FROM pxfm WHERE ascent = 8" 1
# Save As onto another existing file asks to replace; Cancel changes nothing
setdialog save tests/tmp/e2e.pxfdb
key s ctrl shift
wait 6
assert modal Replace_file?
click @Cancel
wait 4
assert modal none
assert wtitle e2e_copy.pxfdb
assert samefile tests/tmp/e2e.pxfdb tests/tmp/e2e_before.pxfdb
# Save As to the file already open is a plain Save
setdialog save tests/tmp/e2e_copy.pxfdb
key s ctrl shift
wait 6
assert modal none
assert wtitle e2e_copy.pxfdb
