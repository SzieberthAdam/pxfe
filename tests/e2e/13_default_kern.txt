# 13. Default glyph and kerning (font 98eae9aa has KERNING A M -1 and a default glyph U+0020): set default, delete drops the default and the pair, clone does not copy kerning.
wait 5
assert kerncount 1
assert default U+0020
click @cell:U+0048
assert selected 1
click @Set_Default
wait 3
assert default U+0048
assert modified 1
key z ctrl
wait 3
assert default U+0020
assert modified 0
click @cell:U+0041
click @cell:U+0048
assert selected 1
assert sel U+0048 1
click @Set_Default
wait 3
assert default U+0048
# deleting the default glyph drops the default
key delete
wait 4
key enter
wait 4
assert glyph U+0048 exists 0
assert default -1
# deleting A drops the kerning pair
click @cell:U+0041
assert sel U+0041 1
key delete
wait 4
key enter
wait 4
assert glyph U+0041 exists 0
assert kerncount 0
# undo brings the pair and the glyph back
key z ctrl
wait 3
assert kerncount 1
# clone of M does not copy kerning
click @cell:U+004D
key d ctrl
wait 4
type U+00E9
key enter
wait 4
assert kerncount 1
