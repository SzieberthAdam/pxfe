# 12. The typeface lock: the typeface is edited in the Typeface view of the library.  Unlocking it makes its fields editable (the italic flag
# is the font's, under the font lock in the Font tab); only the typeface changes (the italic flag, the family), so the typeface is rewritten in place under a new hash (no new pxfm, the same number of pxtf).
wait 5
assert hascont 1
assert readonly 1
assert contlocked 1
assert w.ptab_Typeface 0
click @nav.0
wait 16
assert nav 1
click @bside_Typeface
wait 3
assert w.Unlock 1
# locked: the fields are read-only
click @c.typeface
wait 2
type X
wait 3
assert unsaved 0
assert tabhash ff6354ce745de4d0170855e161a491be
click @tf_lock
wait 12
assert nav 1
assert contlocked 0
assert hascont 1
assert modified 0
assert unsaved 1
assert c.italic 0
snapdir tests/tmp/e2elib
# the italic switch is a field of the font (its variant): the Font tab, under the font's own lock
click @bside_Font
wait 3
click @Unlock
wait 8
assert readonly 0
assert unsaved 1

wait 12
click @c.italic
