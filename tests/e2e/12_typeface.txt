# 12. A typeface duplicate (version 7): the typeface is edited in the Typeface view of the library.  "Duplicate Typeface to Edit" makes an editable
# copy; only the typeface changes (the italic flag, the family), so Save writes a new pxtf (a new hash) and no new pxfm; the original typeface stays.
wait 5
assert hascont 1
assert readonly 1
assert w.ptab_Typeface 0
click @nav.0
wait 16
assert nav 1
assert w.tv.duptypeface 1
click @bside_Typeface
wait 3
assert w.Duplicate_Typeface_to_Edit 1
# an original: the fields are read-only
click @c.family
wait 2
type X
wait 3
assert unsaved 0
assert tabhash b925951bb94de1d15ec27404b3261476
click @Duplicate_Typeface_to_Edit
wait 12
assert nav 1
assert readonly 0
assert hascont 1
assert modified 0
assert unsaved 0
assert w.Duplicate_Typeface_to_Edit 0
assert w.tv.duptypeface 1
assert c.italic 0
snapdir tests/tmp/e2elib
sleep 500
wheel -8 1250 400
wait 12
click @c.italic
wait 10
assert c.italic 1
assert modified 1
assert unsaved 1
assert navseg 0 Test Sans *
sleep 500
wheel -8 1250 400
wait 12
click @c.italic
wait 10
assert c.italic 0
assert modified 0
assert unsaved 0
assert navseg 0 Test Sans
sleep 500
wheel -8 1250 400
wait 12
click @c.italic
wait 10
assert unsaved 1
# the boxed hash is now the copy's, and the original stays shown below it
assert w.hash_typeface_orig 1
assert tabhash !b925951b
sleep 500
wheel 10 1250 400
wait 12
shot tests/tmp/p1/td_9_typeface_edit.png
key s ctrl
wait 12
assert saved pxtf
assert unsaved 0
assert sqlfile tests/tmp/e2e.pxfdb "select count(*) from pxtf" 3
assert sqlfile tests/tmp/e2e.pxfdb "select count(*) from pxfm" 4
assert snapdir unchanged
# the original typeface is still there unchanged, and the copy is a new one
assert sqlfile tests/tmp/e2e.pxfdb "select count(*) from pxtf where lower(hex(hash)) = 'b925951bb94de1d15ec27404b3261476'" 1
assert sqlfile tests/tmp/e2e.pxfdb "select count(*) from pxtf where lower(hex(hash)) <> 'b925951bb94de1d15ec27404b3261476' and lower(hex(hash)) <> 'b5fbcdd0dd5009013337c15e57fa7672'" 1
# a text field: typing changes the copy, Escape puts the value back, backspacing the typed letter returns to the saved state
sleep 500
wheel 10 1250 400
wait 12
click @c.family
type X
key enter
wait 3
assert ctext FAMILY Test SansX
assert unsaved 1
assert navseg 0 Test SansX *
sleep 500
click @c.family

key enter
wait 3

assert unsaved 0
sleep 500
click @c.family
type Y
key escape
wait 3
assert ctext FAMILY Test Sans
assert unsaved 0

sleep 500

type Z

wait 3
