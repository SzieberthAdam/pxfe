# 12. A typeface duplicate: only the typeface changed (italic flag) is enough for Save; a new pxtf is written, the pxfm stays.
wait 5
click @Duplicate_to_Edit
wait 5
assert hascont 1
assert modified 0
assert unsaved 0
assert c.italic 0
snapdir tests/tmp/e2elib
click @ptab_Typeface
wait 3
click @c.italic
wait 3
assert c.italic 1
assert unsaved 1
click @c.italic
wait 3
assert c.italic 0
assert unsaved 0
click @c.italic
wait 3
assert unsaved 1
key s ctrl
wait 12
assert saved pxtf
assert unsaved 0
assert sqlfile tests/tmp/e2e.pxfdb "select count(*) from pxtf" 3
assert sqlfile tests/tmp/e2e.pxfdb "select count(*) from pxfm" 4
assert snapdir unchanged
# never overwrites: the original typeface is still there unchanged
assert sqlfile tests/tmp/e2e.pxfdb "select count(*) from pxtf where lower(hex(hash)) = 'b925951bb94de1d15ec27404b3261476'" 1
