# 109. The Colours switch of the library views: on, only the fonts of more than one colour show (pxfm.colors, NULL counting as 1).
# Database testdata/lib_colors.pxfdb (made by tests\make_testdata.exe): typeface cb07e9fe holds font 3cfdd2b1 (3 colours); typeface 84c74aea
# holds 3b52125b (1 colour) and, italic, c735a3cf (2 colours, no ASCENT); font 894ef821 (2 colours) belongs to no typeface.
wait 10
assert nav 0
assert btab 0
assert bvisible 3
assert bboxes 2
assert bcolors 0

# a typeface shows when any of its fonts has colours, with only those variants
click @Colours
wait 10
assert bcolors 1
assert bvisible 2
assert bboxes 2
shot tests/tmp/shots/109_colours_typefaces.png
click @Colours
wait 6
assert bcolors 0
assert bvisible 3

# the Fonts tab has the same switch: 4 fonts, 3 of them with colours
click @Fonts
wait 10
assert btab 1
assert bvisible 4
click @Colours
wait 10
assert bvisible 3
shot tests/tmp/shots/109_colours_fonts.png
# it combines with the other filters
click @Filter
type 894ef
wait 6
assert bvisible 1
click @Filter_Clear
wait 6
assert bvisible 3

# Clear filters in the empty state turns it off
click @Filter
type zzzz
wait 6
assert bvisible 0
click @Clear_filters
wait 6
assert bcolors 0
assert bvisible 4
assert bfilter -

# the switch stays when the tab changes (pxfe.ini keeps it as colouronly= in [browser] on quitting)
click @Colours
wait 6
assert bcolors 1
click @Typefaces
wait 10
assert btab 0
assert bcolors 1
assert bvisible 2
click @Colours
wait 6
assert bvisible 3
