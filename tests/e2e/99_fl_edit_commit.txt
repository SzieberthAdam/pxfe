# 99. A font with layers edited in place: unlocking keeps the layers, a pixel edit gives the font its own hash, committing moves the layers to
# that hash (nothing stays under the old one), a second layer added after the edit goes with it; Save; the font opens again with both.
# Font 0820dc69 with one outline layer.
wait 5
assert readonly 1
addlayerfile testdata/outline/NotoSub.ttf
wait 6
assert flayers 1
assert fslayers 1
click @Unlock
wait 8
assert readonly 0
assert flayers 1
assert fslayers 1
# a pixel edit (its own hash), then a second layer
click @cell:U+0041
key enter
wait 6
assert editor 1
click @ecell:2,3
wait 4
assert modified 1
assert flayers 1
key escape
wait 6
addlayerfile testdata/bitmap/font.bdf
wait 6
assert flayers 2
assert fslayers 2
wait 6
# leaving the font view commits the font in place: its layers are under its new hash, nothing is left under the old one
click @nav.0
wait 10
assert editpending 0
assert sql "select count(*) from pxfm" 4
assert sqlfile tests/tmp/e2e.pxfdb "select count(*) from pxfm" 4
key s ctrl
wait 12
assert unsaved 0
assert sidecar exists
assert sqlfile tests/tmp/e2e.pxfdb-settings "select count(*) from layer" 2
assert sqlfile tests/tmp/e2e.pxfdb-settings "select count(*) from layer where font = '0820dc69a21b1cdb3a44d2cff7d884d8'" 0
assert sqlfile tests/tmp/e2e.pxfdb-settings "select count(distinct font) from layer" 1
assert sqlfile tests/tmp/e2e.pxfdb "select count(*) from pxfm" 4
# the edited font again: its own single layer
click @Fonts
wait 6
click @Filter
type $NEWPXFM
wait 4
click @card:0
wait 3
dblclick @card:0
wait 8
assert readonly 0
assert hashname 1
assert flayers 2
assert fslayers 2
assert flayer 1 source NotoSub.ttf
assert flayer 1 outline 1
