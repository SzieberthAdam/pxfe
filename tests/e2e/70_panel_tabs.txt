# 70. The panel tabs and their commands (testdata/lib, no font open).
# 1 the library's Font tab: padlock, hash, Clone Font / Delete Font, Export Font and Export Typeface write packages, no right-click menu; 2 Clone Font needs the typeface unlocked, unlocking it from
# the Typeface tab opens its font quietly and stays in the library; 3 Clone Font adds a variant naming the same font; 4 a press on the selected
# entry opens it; 5 the font view's View, Font and Glyph tabs hold what the command bar had; 6 the glyph view's View tab and its sample strip;
# 7 a text field selects (Ctrl+A, then typing replaces the selection).
wait 16
assert nav 0
assert bvisible 3

# ---- 1. the Font tab of a variant row
click @card:0
wait 3
click @bside_Font
wait 3
assert bside 3
assert w.Clone_Font 1
assert w.Delete_Font 1
assert w.Unlock 1
click @card:0 right
wait 3
assert w.Ctx_Open 0
assert w.Ctx_Delete_Font 0
# Export Font... writes a .pxfp package where the dialog says, Export Typeface... a .pxtp
delete tests/tmp/x70.pxfp
delete tests/tmp/x70.pxtp
setdialog save tests/tmp/x70.pxfp
click @Export_Font…
wait 10
assert file tests/tmp/x70.pxfp exists
click @bside_Typeface
wait 3
setdialog save tests/tmp/x70.pxtp
click @Export_Typeface…
wait 10
assert file tests/tmp/x70.pxtp exists
setdialog save none
click @bside_Font
wait 3

# ---- 2. Clone Font is off while the typeface is locked
click @Clone_Font
wait 10
assert bvisible 3
click @bside_Typeface
wait 3
assert w.Delete_Typeface 1
click @tf_lock
wait 12
assert inbrowser 1
assert nav 0
assert hascont 1
assert contlocked 0

# ---- 3. the clone: one more variant, the same fonts, the same number of typefaces
click @bside_Font
wait 3
click @Clone_Font
wait 16
assert bvisible 4
assert storefonts 4
assert storetypefaces 2
assert toast Cloned as variant
