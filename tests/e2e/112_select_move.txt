# 112. Selection and moving (version 16), font 0820dc69 unlocked, glyph U+0020 (empty, advance 4; font box rows 0..7, @ecell:x,y).
# Marquee, drag-move of the selected pixels, drop, undo (one step per move), the arrow keys, Esc, Ctrl+drag copy.
wait 5
click @Unlock
wait 4
sleep 600
dblclick @cell:U+0020
wait 5
assert editor_glyph U+0020
assert glyph U+0020 empty
assert etool 0
assert eselcount 0
assert efloating 0

# a block of 3 x 3 pixels at columns 1..3, rows 1..3
drag @ecell:1,1 @ecell:3,1
drag @ecell:1,2 @ecell:3,2
drag @ecell:1,3 @ecell:3,3
wait 3
assert glyph U+0020 ink 9
assert undo 3

# the marquee tool: key M, the toolbar button, the status line
key m
wait 2
assert etool 5
click @Pencil
wait 2
assert etool 0
click @Marquee
wait 2
assert etool 5
key b
key m
wait 2
assert etool 5

# drag a rectangle: 5 x 5 cells (columns 0..4, rows 0..4), 9 of them hold ink
drag @ecell:0,0 @ecell:4,4
wait 3
assert eselcount 25
assert esel bounds 0 0 5 5
assert esel cell 0 0
assert esel cell 4 4
assert esel cell 5 4 0
assert undo 3
assert dirty 1
