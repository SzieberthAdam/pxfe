# 15. Import Folder (File menu): the testdata/lib copy in tests/tmp/e2elib becomes a new database file, which is then loaded.
# A clean database loaded first: no prompt.  The target exists on the second import: Replace? is asked.
wait 8
assert hasstore 1
delete tests/tmp/e2e_imp.pxfdb
snapdir tests/tmp/e2elib
setdialog folder tests/tmp/e2elib
setdialog save tests/tmp/e2e_imp.pxfdb
click @Menu
wait 3
click @file.import
wait 3
waitfor loading = 0 60000
wait 8
assert modal none
assert hasstore 1
assert storeimported 1
assert inbrowser 1
assert storefonts 4
assert storetypefaces 2
assert storeglyphs 6
assert btotal 3
assert wtitle e2e_imp.pxfdb
assert statusdb e2e_imp.pxfdb
assert statusdb saved
assert file tests/tmp/e2e_imp.pxfdb exists
assert sqlfile tests/tmp/e2e_imp.pxfdb "select count(*) from pxfm" 4
assert sqlfile tests/tmp/e2e_imp.pxfdb "select count(*) from glyph" 6
# the folder is not changed
assert snapdir unchanged
# a second import onto the same target asks to replace; Cancel keeps everything
click @Menu
wait 3
click @file.import
wait 6
assert modal Replace_file?
shot tests/tmp/shots/v5_replace_prompt.png
click @Cancel
wait 4
assert modal none
assert wtitle e2e_imp.pxfdb
click @Menu
wait 3
click @file.import
wait 6
assert modal Replace_file?
click @Replace
waitfor loading = 0 60000
wait 8
assert modal none
assert storeimported 1
assert storefonts 4
assert snapdir unchanged
