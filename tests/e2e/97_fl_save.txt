# 97. Font layers are saved in the sidecar <db>-settings by Save (97b reloads).  Duplicate of 3b52125b without a pixel edit, so the
# layers sit under that hash.  Three layers (outline, bitmap, a copy of an outline file that is deleted at the end), a correction of A.
wait 5
click @Unlock
wait 4
assert sidecar missing
assert fshasany 1
copyfile testdata/outline/NotoSub.ttf tests/tmp/fl/gone.ttf
addlayerfile testdata/outline/NotoSub.ttf
addlayerfile testdata/bitmap/font.bdf
addlayerfile tests/tmp/fl/gone.ttf
assert flayers 3
wait 6
assert unsaved 1
assert fsdirty 1
assert dbmodified 0
assert wtitle modified
# a settings-only change counts as a modification (unsaved, status bar), the store itself is unchanged
assert statusdb modified
click @cell:U+0041
key enter
wait 6
click @gtab_Layers
wait 3
# the stack: gone.ttf (top, selected), font.bdf, NotoSub.ttf; set up the top one, then the outline one
click @lay_hex
type #123456
key enter
wait 3
click @lay_gx+
click @lay_gx+
click @lay_gy-
wait 3
click @lay_ghide
wait 3
assert fcorr 0 U+0041 hidden 1
click @lay_row_2
wait 3
click @lay_x+
click @lay_x+
click @lay_x+
wait 3
click @lay_gsize.value
type 110
key enter
wait 3
assert fcorr 2 U+0041 dsize 1.1

wait 6
assert flayer 2 dx 0.75
assert flayer 0 color 0x123456
assert flayer 0 ncorr 1
assert fcorr 0 U+0041 dx 0.5
assert fcorr 0 U+0041 dy -0.25
# Save writes the sidecar and nothing else
copyfile tests/tmp/e2e.pxfdb tests/tmp/e2e_before.pxfdb
